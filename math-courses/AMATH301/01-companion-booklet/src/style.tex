\newcommand{\chapterbannerbase}[3]{%
\begin{tikzpicture}[remember picture,overlay]
  \coordinate (A) at (current page.north west);
  \coordinate (B) at ([yshift=-6.3cm]current page.north east);
  \fill[plum] (A) rectangle (B);
  \begin{scope}
    \clip (A) rectangle (B);
    \draw[white, opacity=0.07, line width=0.3pt, step=5mm]
      ([yshift=-7cm]current page.north west) grid ([yshift=1cm]current page.north east);
    \draw[white, opacity=0.14, line width=0.5pt, step=2.5cm]
      ([yshift=-7cm]current page.north west) grid ([yshift=1cm]current page.north east);
    \foreach \r in {0.6,1.2,...,6.0}
      \draw[coral, opacity=0.22, line width=0.6pt]
        ([xshift=-3.2cm,yshift=-3.6cm]current page.north east) circle (\r);
  \end{scope}
  \fill[coral] ([yshift=-6.3cm]current page.north west) rectangle
               ([yshift=-6.55cm]current page.north east);
  \node[anchor=north west, inner sep=0, text=coral,
        font=\fontsize{64}{64}\selectfont\bfseries]
    at ([xshift=2.1cm,yshift=-1.25cm]current page.north west) {#1};
  \node[anchor=north west, inner sep=0, text=white!80!plum, font=\small\scshape]
    at ([xshift=#3,yshift=-1.45cm]current page.north west) {#2};
  \node[anchor=north west, inner sep=0, text=white, text width=13.6cm,
        font=\fontsize{24}{29}\selectfont\scshape, align=left]
    at ([xshift=#3,yshift=-2.15cm]current page.north west)
    {\raggedright\hyphenchar\font=-1 \bannertitle};
\end{tikzpicture}\vspace*{3.9cm}}
